<script>
const KEY = "نجوم_القراءة";
let students = JSON.parse(localStorage.getItem(KEY)) || [];

function save() {
  localStorage.setItem(KEY, JSON.stringify(students));
}

function addStudent() {
  const input = document.getElementById("studentName");
  const name = input.value.trim();

  if (!name) {
    alert("اكتبي اسم الطالب أولاً 🌷");
    return;
  }

  students.push({
    id: Date.now(),
    name: name,
    stars: 0
  });

  save();
  input.value = "";
  render();
}

function addStar(id) {
  const student = students.find(s => s.id === id);
  if (!student) return;

  student.stars++;
  save();
  render();
  celebrate(student);
}

function getLevel(stars) {
  if (stars >= 20) return "🏆 بطل القراءة";
  if (stars >= 15) return "👑 قارئ متميز";
  if (stars >= 10) return "🌟 قارئ متألق";
  if (stars >= 5) return "📚 قارئ مجتهد";
  return "🌱 قارئ واعد";
}

function makeStars(number) {
  if (number === 0) return "✨ بانتظار أول نجمة";

  let result = "";
  const shown = Math.min(number, 20);

  for (let i = 0; i < shown; i++) {
    result += "⭐";
  }

  if (number > 20) {
    result += " +" + (number - 20);
  }

  return result;
}

function render() {
  const area = document.getElementById("students");
  area.innerHTML = "";

  students.forEach(student => {
    const card = document.createElement("div");
    card.className =
      "student-card" +
      (student.stars >= 20 ? " champion" : "");

    const name = document.createElement("div");
    name.className = "student-name";
    name.textContent = student.name;

    const count = document.createElement("div");
    count.className = "star-count";
    count.textContent = student.stars + " ⭐";

    const stars = document.createElement("div");
    stars.className = "stars";
    stars.textContent = makeStars(student.stars);

    const button = document.createElement("button");
    button.className = "star-btn";
    button.textContent = "⭐ أضف نجمة";
    button.onclick = () => addStar(student.id);

    const deleteButton = document.createElement("button");
    deleteButton.className = "delete-btn";
    deleteButton.textContent = "🗑️ حذف الطالب";
    deleteButton.onclick = () => deleteStudent(student.id);

    const level = document.createElement("div");
    level.className = "level";
    level.textContent = getLevel(student.stars);

    card.appendChild(name);
    card.appendChild(count);
    card.appendChild(stars);
    card.appendChild(button);
    card.appendChild(deleteButton);
    card.appendChild(level);

    area.appendChild(card);
  });
}

function celebrate(student) {
  const box = document.getElementById("celebration");
  const text = document.getElementById("celebrationText");

  if (student.stars >= 20) {
    text.textContent =
      "🏆 أحسنت يا " + student.name +
      "! أنت بطل القراءة";
  } else {
    text.textContent =
      "أحسنت يا " + student.name +
      " 🌟 نجمة جديدة!";
  }

  box.classList.add("show");

  setTimeout(() => {
    box.classList.remove("show");
  }, 1300);
}

function deleteStudent(id) {
  const student = students.find(s => s.id === id);
  if (!student) return;

  const ok = confirm(
    "هل أنتِ متأكدة من حذف " + student.name + "؟"
  );

  if (ok) {
    students = students.filter(s => s.id !== id);
    save();
    render();
  }
}

function exportData() {
  const data = JSON.stringify(students, null, 2);
  const blob = new Blob([data], {
    type: "application/json"
  });

  const url = URL.createObjectURL(blob);
  const a = document.createElement("a");

  a.href = url;
  a.download = "نجوم-القراءة-نسخة-احتياطية.json";
  a.click();

  setTimeout(() => {
    URL.revokeObjectURL(url);
  }, 1000);
}

function importData(event) {
  const file = event.target.files[0];
  if (!file) return;

  const reader = new FileReader();

  reader.onload = function(e) {
    try {
      const imported = JSON.parse(e.target.result);

      if (!Array.isArray(imported)) {
        throw new Error();
      }

      const ok = confirm(
        "سيتم استبدال البيانات الحالية بالنسخة المحفوظة. هل تريدين المتابعة؟"
      );

      if (!ok) {
        event.target.value = "";
        return;
      }

      students = imported.map(s => ({
        id: Number(s.id) || Date.now() + Math.random(),
        name: String(s.name || "طالب"),
        stars: Math.max(0, Number(s.stars) || 0)
      }));

      save();
      render();

      alert("تمت استعادة النسخة بنجاح ⭐");
    } catch {
      alert("تعذر قراءة ملف النسخة الاحتياطية");
    }

    event.target.value = "";
  };

  reader.readAsText(file);
}

render();
